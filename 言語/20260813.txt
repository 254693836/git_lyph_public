Sub Macro2()
With ActiveSheet

    '先にフィルターをクリア
    If .AutoFilterMode Then
        If .FilterMode Then .ShowAllData
    End If

    'C1の「、」で区切られた検索語を配列にする
    Dim arr As Variant
    arr = Split(.Range("C1").Value, "、")

    '各検索語を曖昧検索
    Dim i As Long
    For i = LBound(arr) To UBound(arr)
        arr(i) = "*" & Trim(arr(i)) & "*"
    Next i

    .Range("$A$2:$H$853").AutoFilter _
        Field:=3, _
        Criteria1:=arr, _
        Operator:=xlFilterValues

End With
    
End Sub
